# ─────────────────────────────────────────────────────────────
#  GLOBAL SEARCH — Journal Entry
# ─────────────────────────────────────────────────────────────

@frappe.whitelist()
def search_journal_entry(doctype, txt, searchfield, start, page_len, filters):
    """
    Custom search handler for Journal Entry in global search.
    Searches by name, reference_number, and remarks.
    """
    filters = frappe.parse_json(filters) if isinstance(filters, str) else (filters or {})
    
    or_filters = []
    if txt:
        or_filters.append(["name", "like", f"%{txt}%"])
        or_filters.append(["reference_number", "like", f"%{txt}%"])
        or_filters.append(["remarks", "like", f"%{txt}%"])
    
    records = frappe.get_list(
        "Journal Entry",
        filters={"docstatus": 1},  # Show only submitted entries
        or_filters=or_filters or None,
        fields=["name", "posting_date", "remarks"],
        start=start,
        page_length=page_len,
        order_by="posting_date desc"
    )
    
    return [[r.name, f"{r.posting_date} - {r.remarks or ''}"] for r in records]
```

function validate_unique_sections_in_form(frm) {
    const seen = new Map();

    function track(section, source, idx) {
        const key = (section || "").trim();
        if (!key) return;

        const first = seen.get(key);
        if (first) {
            frappe.throw(
                __(
                    "Duplicate section name <b>{0}</b> found in this setting.<br><br>First used in: {1}<br>Duplicate at: {2}",
                    [
                        frappe.utils.escape_html(key),
                        frappe.utils.escape_html(first),
                        frappe.utils.escape_html(`${source} row ${idx}`),
                    ]
                )
            );
        }

        seen.set(key, `${source} row ${idx}`);
    }

    (frm.doc.match_criteria || []).forEach((row, i) => {
        track(row.section, "Match Criteria", row.idx || i + 1);
    });

    (frm.doc.approvers || []).forEach((row, i) => {
        track(row.section, "Approvers", row.idx || i + 1);
    });
}
```

#### 2. ENHANCED: Section Matching Logic
- **Added Default Section Handling**:
```python
default_section = (setting.get("default_section") or "").strip()

for sec in all_sections:
    # Default section is fallback-only — its own criteria are ignored
    # and it never wins through the normal scoring loop.
    if sec == default_section:
        continue

    criteria = criteria_by_section.get(sec, [])
    # A non-default section with zero criteria matches nothing —
    # fallback behavior is reserved for the explicit default_section.
    if not criteria:
        continue

    if all(
        str(doc.get(c.field_name) or "").strip() == str(c.field_value or "").strip()
        for c in criteria
    ):
        score = len(criteria)
        if score > best_score:
            best_score = score
            best_section = sec

# No criteria match? fall back to configured default section.
if best_section is None and default_section and default_section in all_sections:
    best_section = default_section
```
